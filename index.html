<script>
let userState = {
balance: 100,
referCode: "FLAX" + Math.floor(1000 + Math.random() * 9000)
};

let matchesData = [
{ id: 1, title: "BGMI Erangel Special", type: "SOLO", map: "Erangel", fee: 20, kill: 15, prize: 500 },
{ id: 2, title: "Free Fire Clash Squad", type: "SQUAD", map: "Bermuda", fee: 30, kill: 20, prize: 800 }
];

let leaderboardData = [
{ name: "ProGamer_99", earnings: "₹4,500" },
{ name: "Alpha_King", earnings: "₹3,200" },
{ name: "Flax_Master", earnings: "₹2,800" }
];

document.addEventListener("DOMContentLoaded", () => {
updateUI();
renderMatches();
renderLeaderboard();
});

function switchTab(tabId) {
document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));

document.getElementById(tabId).classList.add('active');
event.currentTarget.classList.add('active');
}

function updateUI() {
document.getElementById('navBalance').innerText = userState.balance;
document.getElementById('walletBalance').innerText = userState.balance;
document.getElementById('userReferCode').innerText = userState.referCode;
}

function renderMatches() {
const list = document.getElementById('matchesList');
list.innerHTML = "";
matchesData.forEach(m => {
list.innerHTML += `
<div class="card">
<div class="match-header">
<strong style="font-size: 15px;">${m.title}</strong>
<span class="match-type">${m.type}</span>
</div>
<div class="match-map">Map: ${m.map}</div>
<div class="match-details" style="margin-top: 10px;">
<div class="detail-item"><span>Entry</span><strong>₹${m.fee}</strong></div>
<div class="detail-item"><span>Per Kill</span><strong>₹${m.kill}</strong></div>
<div class="detail-item"><span>Prize Pool</span><strong>₹${m.prize}</strong></div>
</div>
<button class="join-btn" onclick="joinMatch(${m.fee})">JOIN MATCH</button>
</div>
`;
});
}

function renderLeaderboard() {
const list = document.getElementById('leaderboardList');
list.innerHTML = "";
leaderboardData.forEach((item, index) => {
let rankClass = index < 3 ? `rank-${index+1}` : '';
list.innerHTML += `
<div class="leaderboard-item">
<div style="display: flex; align-items: center; gap: 10px;">
<span class="rank ${rankClass}">${index + 1}</span>
<strong style="font-size: 14px;">${item.name}</strong>
</div>
<span style="color: #28a745; font-weight: 700;">${item.earnings}</span>
</div>
`;
});
}

function joinMatch(fee) {
if (userState.balance >= fee) {
userState.balance -= fee;
updateUI();
alert("Successfully Joined Match!");
} else {
alert("Insufficient Balance! Please add money to your wallet.");
switchTab('wallet');
}
}

function handleDeposit(e) {
e.preventDefault();
alert("Deposit Request Submitted! Pending Admin Approval.");
e.target.reset();
}

function handleWithdraw(e) {
e.preventDefault();
const amt = parseInt(document.getElementById('withdrawAmount').value);
if (amt <= userState.balance) {
userState.balance -= amt;
updateUI();
alert("Withdrawal Request Submitted!");
e.target.reset();
} else {
alert("Insufficient Balance!");
}
}

function copyReferCode() {
navigator.clipboard.writeText(userState.referCode);
alert("Referral code copied!");
}

function updateQr() {
const url = document.getElementById('adminQrInput').value;
if(url) {
document.getElementById('displayQr').src = url;
alert("QR Code updated!");
}
}

function createMatch(e) {
e.preventDefault();
const newM = {
id: Date.now(),
title: document.getElementById('mTitle').value,
type: "SOLO",
map: document.getElementById('mMap').value,
fee: document.getElementById('mFee').value,
kill: document.getElementById('mKill').value,
prize: document.getElementById('mPrize').value
};
matchesData.push(newM);
renderMatches();
alert("New Match Created!");
e.target.reset();
switchTab('matches');
}
</script>
